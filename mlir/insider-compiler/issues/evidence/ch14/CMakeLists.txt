cmake_minimum_required(VERSION 3.20)
project(insider_ch14_evidence LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
add_executable(ch14-analysis analysis.cpp)
target_compile_features(ch14-analysis PRIVATE cxx_std_17)
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(ch14-analysis PRIVATE -fno-rtti)
endif()
target_include_directories(ch14-analysis PRIVATE ${MLIR_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
target_compile_definitions(ch14-analysis PRIVATE ${LLVM_DEFINITIONS})
target_link_libraries(ch14-analysis PRIVATE MLIRAffineAnalysis MLIRAffineUtils MLIRParser MLIRFuncDialect MLIRMemRefDialect MLIRArithDialect)
