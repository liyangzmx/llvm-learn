cmake_minimum_required(VERSION 3.20)
project(insider_ch11_evidence LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
add_executable(ch11-translate translate.cpp)
target_compile_features(ch11-translate PRIVATE cxx_std_17)
target_include_directories(ch11-translate PRIVATE ${MLIR_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
target_compile_definitions(ch11-translate PRIVATE ${LLVM_DEFINITIONS})
target_link_libraries(ch11-translate PRIVATE MLIRLLVMToLLVMIRTranslation MLIRBuiltinToLLVMIRTranslation MLIRParser)
add_executable(ch11-types types.cpp)
target_compile_features(ch11-types PRIVATE cxx_std_17)
target_include_directories(ch11-types PRIVATE ${MLIR_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
target_compile_definitions(ch11-types PRIVATE ${LLVM_DEFINITIONS})
target_link_libraries(ch11-types PRIVATE MLIRLLVMCommonConversion MLIRAsmParser)
