cmake_minimum_required(VERSION 3.20)
project(insider_ch15_evidence LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
add_executable(ch15-check check.cpp)
target_compile_features(ch15-check PRIVATE cxx_std_17)
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(ch15-check PRIVATE -fno-rtti)
endif()
target_include_directories(ch15-check PRIVATE ${MLIR_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
target_compile_definitions(ch15-check PRIVATE ${LLVM_DEFINITIONS})
target_link_libraries(ch15-check PRIVATE MLIRPresburger)
